def eq(a, b, label):
    counters['assertions'] += 1
    if isinstance(a, bool) or isinstance(b, bool) or a is None or b is None:
        ok = a == b
    elif isinstance(a, (int,float)) and isinstance(b, (int,float)):
        ok = abs(a-b) <= 1e-10
    else:
        ok = a == b
    if not ok:
        failures.append((label, a, b))

def compare_arithmetic(values, actual, label):
    D, V, Lh, dh = [values[k] for k in ('D','V','Lh','dh')]
    if None in (D,V,Lh,dh):
        raise AssertionError(('incomplete',label,values))
    theta = V * math.pi / 180
    horizontal, vertical = D*math.cos(theta), D*math.sin(theta)
    hres, vres = Lh-horizontal, dh-vertical
    expected = {'calculated_Lh':horizontal,'calculated_dh':vertical,
                'norm_distance_from_aux':math.sqrt(Lh*Lh+dh*dh),
                'norm_residual_m':math.sqrt(Lh*Lh+dh*dh)-D,
                'inclination_from_aux_deg':math.atan2(dh,Lh)*180/math.pi,
                'Lh_equals_D_at_nonzero_V':Lh==D and V!=0,
                'azimuth_testable':False}
    eq(actual['status'],'complete',label+'/status')
    for k,v in expected.items():eq(actual[k],v,label+'/'+k)
    for f,residual in [('Lh',hres),('dh',vres)]:
        eq(actual['residual_scan_minus_calculated'][f],residual,label+'/residual/'+f)
        eq(actual['strict_rounding_0_005m'][f],abs(residual)<=.005,label+'/strict/'+f)
        eq(actual['within_0_05m'][f],abs(residual)<=.05,label+'/review/'+f)
    counters['arithmetic_records'] += 1

def normalize(s):return s.replace('−','-').replace('\\','').replace(',','.')

def parse_cell(s):
    vals=[float(n) for n in re.findall(r'[-+]?\d+(?:\.\d+)?',re.sub(r'([+-])\s+(?=\d)', r'\1', normalize(s)))]
    return (vals[0] if vals else None),list(dict.fromkeys(vals[1:]))

def parse_md(text):
    out=[]
    for line_no,line in enumerate(text.splitlines(),1):
        if not line.startswith('|'):continue
        cells=[c.strip() for c in line.strip('|').split('|')]
        if len(cells)<7 or not all(re.fullmatch(r'(?:\d+|a|b)',c) for c in cells[:2]):continue
        parsed={k:parse_cell(cells[n]) for k,n in [('D',4),('A',2),('V',3),('Lh',5),('dh',6)]}
        out.append({'from':cells[0],'to':cells[1],'line':line_no,'raw':line,
                    'values':{k:v[0] for k,v in parsed.items()},'alternatives':{k:v[1] for k,v in parsed.items()}})
    return out

def parse_srv(text):
    out=[]; pending={}
    for line_no,line in enumerate(text.splitlines(),1):
        if line.startswith('; DO_WERYFIKACJI:'):
            m=re.search(r'([DAV])=([+-]?[0-9.]+).*?(?:odczyt |alternatywa |pozostaje )([+-]?[0-9.]+)',line)
            if m:pending={m[1]:[float(m[3].rstrip('.'))]}
            else:
                m=re.search(r'([DAV])=([+-]?[0-9.]+).*?([+-]?[0-9.]+)\.',line)
                if m:pending={m[1]:[float(m[3].rstrip('.'))]}
        if not re.match(r'^(?:\d+|a)\s+(?:\d+|a|b)\s+',line):continue
        main,comment=line.split(';',1);cells=main.split()
        vals={'D':float(cells[2]),'A':None if cells[3]=='--' else float(cells[3]),'V':float(cells[4])}
        for field,pat in [('Lh',r'Lh_scan=([+\-\d.]+)'),('dh',r'dh_scan=([+\-\d.]+)')]:vals[field]=float(re.search(pat,comment)[1])
        out.append({'from':cells[0],'to':cells[1],'line':line_no,'raw':line,'values':vals,'alternatives':pending})
        pending={}
    return out

def independent_comparison(rows):
    unique={};repeat_conflicts=[]
    for r in rows:
        key=(r['from'],r['to'])
        if key not in unique:unique[key]=r
        elif unique[key]['values']!=r['values']:repeat_conflicts.append(key)
    counts=collections.Counter();by_field={f:collections.Counter() for f in FIELDS};mismatches=[]
    for key,ref in refs.items():
        for f in FIELDS:
            if f in ref['uncertain_fields']:continue
            expected=ref['values'][f];counts['assessed']+=1;by_field[f]['assessed']+=1
            row=unique.get(key)
            if row is None:
                counts['missing']+=1;by_field[f]['missing']+=1;mismatches.append((key,f,None,expected,'missing_pair'));continue
            primary=row['values'][f]
            if primary==expected:
                counts['correct']+=1;by_field[f]['correct']+=1
            else:
                counts['wrong']+=1;by_field[f]['wrong']+=1;mismatches.append((key,f,primary,expected,'different'))
            if primary==expected or expected in row['alternatives'].get(f,[]):
                counts['including_alternatives']+=1;by_field[f]['including_alternatives']+=1
    return counts,by_field,mismatches,unique,repeat_conflicts

def path_to(station):
    path=[];seen=set()
    while station!='0':
        if station in seen:raise ValueError('cycle')
        seen.add(station);row=parents[station];path.append(row);station=row['from']
    return list(reversed(path))
def path_sums(rows):
    scan=sum(r['values']['dh'] for r in rows)
    trig=sum(r['values']['D']*math.sin(r['values']['V']*math.pi/180) for r in rows)
    choices=[[r['values']['dh']]+r['alternatives'].get('dh',[]) for r in rows]
    return scan,trig,[sum(min(x) for x in choices),sum(max(x) for x in choices)]
