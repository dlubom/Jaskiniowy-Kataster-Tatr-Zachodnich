cmake_minimum_required(VERSION 3.20)
project(walls_driver C)
add_executable(walls-driver driver.c)
set_property(TARGET walls-driver PROPERTY C_STANDARD 11)
target_link_libraries(walls-driver PRIVATE user32)
if(MSVC)
  target_compile_definitions(walls-driver PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(walls-driver PRIVATE /W4 /WX)
else()
  target_compile_options(walls-driver PRIVATE -Wall -Wextra -Werror)
endif()
